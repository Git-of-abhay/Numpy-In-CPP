cmake_minimum_required(VERSION 3.16)
project(numcpp_v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(numcpp Numc++.cpp)
target_include_directories(numcpp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(numcpp_tests tests/test_array.cpp)
target_link_libraries(numcpp_tests PRIVATE numcpp)
add_executable(numcpp_statistics_tests tests/test_statistics.cpp)
target_link_libraries(numcpp_statistics_tests PRIVATE numcpp)
enable_testing()
add_test(NAME numcpp_tests COMMAND numcpp_tests)
add_test(NAME numcpp_statistics_tests COMMAND numcpp_statistics_tests)
